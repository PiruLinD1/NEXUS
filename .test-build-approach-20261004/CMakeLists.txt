cmake_minimum_required(VERSION 3.20)
project(approach_probe LANGUAGES CXX)
foreach(version before after)
  if(version STREQUAL "before")
    set(controller before/controller.cpp)
  else()
    set(controller ../src/nexus/controller.cpp)
  endif()
  add_executable(approach_${version} benchmark.cpp ${controller})
  target_compile_features(approach_${version} PRIVATE cxx_std_23)
  if(version STREQUAL "before")
    target_include_directories(approach_${version} PRIVATE before/include)
  endif()
  target_include_directories(approach_${version} PRIVATE ../include)
endforeach()
