cmake_minimum_required(VERSION 3.20)
project(arrival_comparison LANGUAGES CXX)
foreach(version before after)
  if(version STREQUAL "before")
    set(controller controller-before.cpp)
  else()
    set(controller ../src/nexus/controller.cpp)
  endif()
  add_executable(arrival_${version} benchmark.cpp ${controller})
  target_compile_features(arrival_${version} PRIVATE cxx_std_23)
  target_include_directories(arrival_${version} PRIVATE ../include)
endforeach()
