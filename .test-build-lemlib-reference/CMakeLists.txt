cmake_minimum_required(VERSION 3.20)
project(lemlib_reference_audit LANGUAGES CXX)
enable_testing()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/odom.cpp" ODOM_SHA)
if(NOT ODOM_SHA STREQUAL "26a84c6a3448c0443a398fef437ae6597d44cb3d0d5485748c3f4604005df5f6")
    message(FATAL_ERROR "LemLib odom.cpp differs from pinned v0.5.6 bytes")
endif()
add_executable(lemlib_compare compare.cpp odom.cpp ../src/nexus/estimator.cpp)
target_compile_features(lemlib_compare PRIVATE cxx_std_23)
target_include_directories(lemlib_compare BEFORE PRIVATE fake ../include)
add_test(NAME lemlib_differential COMMAND lemlib_compare)
add_test(NAME lemlib_native_differential COMMAND lemlib_compare --native)
