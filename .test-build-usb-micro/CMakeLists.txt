cmake_minimum_required(VERSION 3.20)
project(usb_pulse_guard LANGUAGES CXX)
add_executable(guard_test guard_test.cpp)
target_compile_features(guard_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(guard_test PRIVATE /UNDEBUG)
else()
  target_compile_options(guard_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME pulse_guard COMMAND guard_test)
